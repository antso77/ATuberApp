Screen1.Initialize
    set CustomWebView1.JavaScriptEnabled to true
    set CustomWebView1.DomStorageEnabled to true
    set CustomWebView1.FullscreenVideo to true
    call CustomWebView1.GoToUrl(
        "https://KAYTTAJANIMI.github.io/REPOSITORY/"
    )
